cmake_minimum_required(VERSION 3.16)
project(memo_whisper_worker LANGUAGES C CXX)
if(NOT MEMO_WHISPER_SOURCE)
  message(FATAL_ERROR "MEMO_WHISPER_SOURCE must point to the pinned whisper.cpp checkout")
endif()
add_subdirectory("${MEMO_WHISPER_SOURCE}" whisper)
add_executable(memo-whisper-worker whisper-worker.cpp)
set_target_properties(memo-whisper-worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_link_libraries(memo-whisper-worker PRIVATE whisper)
target_compile_features(memo-whisper-worker PRIVATE cxx_std_17)

# Windows streaming Nemotron adapter, using NVIDIA's pinned CPU-only C ABI SDK.
if(MEMO_NEMO_SDK)
  add_executable(memo-nemotron-worker ../../sidecars/dictation/nemotron-worker.cpp)
  target_include_directories(memo-nemotron-worker PRIVATE "${MEMO_NEMO_SDK}/include")
  target_link_libraries(memo-nemotron-worker PRIVATE "${MEMO_NEMO_SDK}/lib/nemo_speech_asr_c.lib")
  target_compile_features(memo-nemotron-worker PRIVATE cxx_std_17)
  set_target_properties(memo-nemotron-worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
